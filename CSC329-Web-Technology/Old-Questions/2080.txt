1. js validation
<html>
<head>
    <title>Form Validation</title>
    <script>
        function validateForm()
        {
            var name = document.userForm.nameField.value;
            var country = document.userForm.countryField.value;
            var gender = document.userForm.genderField;
            var password = document.userForm.passwordField.value;
       
            const namePattern = /^.{4,}$/;

            const passwordPattern = /^[A-Za-z].*#$/;

            if(name == "" || !namePattern.test(name))
            {
                alert("Name is required, must contain letters and at least 4 characters");
                return false;
            }

            if(country == "")
            {
                alert("Country can't be empty");
                return false;
            }

            var genderSelected = false;
            for(var i = 0; i < gender.length; i++)
            {
                if(gender[i].checked)
                {
                    genderSelected = true;
                    break;
                }
            }
            if(!genderSelected)
            {
                alert("Please select a gender");
                return false;
            }

            if(password == "" || !passwordPattern.test(password))
            {
                alert("Password is required, must start with character and end with #");
                return false;
            }

            alert("Validation successful");
            return true;
        }
    </script>
</head>
<body>
    <form name="userForm" onsubmit="return validateForm()" method="post">

3. php data insert and retrieve
<?php
// Database connection parameters (assumptions)
$servername = "localhost";
$username = "root";
$password = "";
$dbname = "collegeDB";

// 1. Create connection
$conn = new mysqli($servername, $username, $password, $dbname);

// Check connection
if ($conn->connect_error) {
  die("Connection failed: " . $conn->connect_error);
}

// 2. INSERTING values
$sql_insert = "INSERT INTO Students (name, email) VALUES ('Sita', 'sita@example.com')";
if ($conn->query($sql_insert) === TRUE) {
  echo "New record created successfully<br>";
} else {
  echo "Error: " . $conn->error;
}

// 3. SELECTING values
$sql_select = "SELECT name, email FROM Students";
$result = $conn->query($sql_select);

if ($result->num_rows > 0) {
  // Output data of each row
  while($row = $result->fetch_assoc()) {
    echo "Name: " . $row["name"]. " - Email: " . $row["email"]. "<br>";
  }
} else {
  echo "0 results";
}

// 4. Close connection
$conn->close();
?>

7. position property
-specifies type of positioning method used for element
-set positioin values then position useing tblr propoerties

static:
-deafult positionaing behaviour of element according to normal documeent flow
-ignores offset properties like tblr
-not affected by z index property

fixed:
-position element relative to browser window/viewport regardless of scrolling
-removed from normal docuemnt flow
-desont affect positioning of other elemetns, gets treated as invisble
-position is determined by offset properties tbrl in relation to viewport
-can overlap other elemnts on page, highset z index will be displayed on top
-used for sticky headers,sidebars,navigation menus

sticky:
-beahvse as relative within parent container until specified threshold is reached by scorlling
after which becomes fixed within viewport until scrolling reaches another specified point

relative:
-positioned relative to normal position, remain in normal flow of document
-other elemnts will not be adjusted to fit in gap left by relative elemnet
-can be layered using z index property
-when child is set to absolute with parent relative, child is positined relative to parent
instead of viewport, used for relative positionnng hierarchy

absolute:
-positioned relative to nearest positioned ancestor
-when child is set to absolute with parent relative, child is positined relative to parent
instead of viewport, used for relative positionnng hierarchy
-if no positioned ancestors, uses document body and moves along with page scrolling
-removee from normal flow and can overlap elemetns

10. XMLHttpRequest object
-core of AJAX, used to exchange data with a server in the background without reloading the page

function loadData() {
  var xhr = new XMLHttpRequest(); // 1. Create object
  xhr.open("GET", "data.txt", true); // 2. Initialize request
  xhr.onreadystatechange = function() { // 3. Handle response
    if (xhr.readyState == 4 && xhr.status == 200) {
      document.getElementById("content").innerHTML = xhr.responseText;
    }
  };
  xhr.send(); // 4. Send request
}

12. php session
-stores information on the server to be used across multiple pages for a single user

    Start: Must use session_start() at the very top of the PHP script.
    Set: Store data using the $_SESSION superglobal (e.g., $_SESSION["user"] = "Ram";).
    Access: Retrieve the value on any page by accessing $_SESSION["user"].
    Destroy: End the session using session_destroy().
